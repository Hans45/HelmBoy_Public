34void OpenGLBackground::init(OpenGLContext& open_gl_context) {
35 gl_ready_ = false;
36 position_.reset();
37 texture_coordinates_.reset();
38 texture_uniform_.reset();
39 static const float vertices[] = {
40 -1.0f, 1.0f, 0.0f, 1.0f,
41 -1.0f, -1.0f, 0.0f, 0.0f,
42 1.0f, -1.0f, 1.0f, 0.0f,
43 1.0f, 1.0f, 1.0f, 1.0f
44 };
45
46 memcpy(vertices_, vertices, 16 * sizeof(float));
47
48 static const int triangles[] = {
49 0, 1, 2,
50 2, 3, 0
51 };
52
53 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
54 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
55
56 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(sizeof(vertices)));
57 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
58 vertices_, GL_STATIC_DRAW);
59
60 open_gl_context.extensions.glGenBuffers(1, &triangle_buffer_);
61 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
62
63 GLsizeiptr tri_size = static_cast<GLsizeiptr>(static_cast<size_t>(sizeof(triangles)));
64 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, tri_size,
65 triangles, GL_STATIC_DRAW);
66
67 const char* vertex_shader = Shaders::getShader(Shaders::kBackgroundImageVertex);
68 const char* fragment_shader = Shaders::getShader(Shaders::kBackgroundImageFragment);
69
70 image_shader_ = std::make_unique<OpenGLShaderProgram>(open_gl_context);
71
72 if (image_shader_->addVertexShader(OpenGLHelpers::translateVertexShaderToV3(vertex_shader)) &&
73 image_shader_->addFragmentShader(OpenGLHelpers::translateFragmentShaderToV3(fragment_shader)) &&
74 image_shader_->link()) {
75 image_shader_->use();
76 position_ = std::make_unique<OpenGLShaderProgram::Attribute>(*image_shader_, "position");
77 texture_coordinates_ = std::make_unique<OpenGLShaderProgram::Attribute>(*image_shader_, "tex_coord_in");
78 texture_uniform_ = std::make_unique<OpenGLShaderProgram::Uniform>(*image_shader_, "image");
79 gl_ready_ = vertex_buffer_ != 0 && triangle_buffer_ != 0 &&
80 position_->attributeID != static_cast<GLuint>(-1) &&
81 texture_coordinates_->attributeID != static_cast<GLuint>(-1) &&
82 texture_uniform_->uniformID >= 0;
83 }
84 if (!gl_ready_)
85 Logger::writeToLog("[OpenGL] Background initialization failed: " + image_shader_->getLastError());
86}

88void OpenGLBackground::destroy(OpenGLContext& open_gl_context) {

96 texture_uniform_ = nullptr;

99 open_gl_context.extensions.glDeleteBuffers(1, &triangle_buffer_);
100 vertex_buffer_ = 0;
101 triangle_buffer_ = 0;
102}

133void OpenGLBackground::render(OpenGLContext& open_gl_context) {
134 if (!gl_ready_)
135 return;
136
137 MOPO_ASSERT(glGetError() == GL_NO_ERROR);
138 open_gl_context.extensions.glActiveTexture(GL_TEXTURE0);
139
140 bool background_changed = false;
141 {
142 const ScopedLock lock(background_lock_);
143 if (new_background_) {
144 background_image_ = pending_background_image_;
145 new_background_ = false;
146 background_changed = true;
147 }
148 }
149
150 if ((background_changed || background_.getWidth() == 0) && background_image_.getWidth() > 0) {
151 background_.loadImage(background_image_);
152
153 float width_ratio = (1.0f * background_.getWidth()) / background_image_.getWidth();
154 float height_ratio = (1.0f * background_.getHeight()) / background_image_.getHeight();
155 float width_end = 2.0f * width_ratio - 1.0f;
156 float height_end = 1.0f - 2.0f * height_ratio;
157
158 vertices_[8] = vertices_[12] = width_end;
159 vertices_[5] = vertices_[9] = height_end;
160
161 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
162 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
163 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
164 vertices_, GL_STATIC_DRAW);
165 }
166
167 image_shader_->use();
168 bind(open_gl_context);
169 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
170 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
171
172 if (texture_uniform_ != nullptr && background_.getWidth())
173 texture_uniform_->set(0);
174
175 enableAttributes(open_gl_context);
176 glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);
177 disableAttributes(open_gl_context);
178 background_.unbind();
179
180 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
181 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
182 MOPO_ASSERT(glGetError() == GL_NO_ERROR);
183}
