60OpenGLOscilloscope::OpenGLOscilloscope() : output_memory_(nullptr) {
61 line_data_ = new float[2 * RESOLUTION];
62 line_indices_ = new int[2 * RESOLUTION];
63
64 for (int i = 0; i < RESOLUTION; ++i) {
65 float t = i / (RESOLUTION - 1.0f);
66 line_data_[2 * i] = 2.0f * t - 1.0f;
67 line_data_[2 * i + 1] = 0.0f;
68
69 line_indices_[2 * i] = i;
70 line_indices_[2 * i + 1] = i + 1;
71 }
72
73 line_indices_[2 * RESOLUTION - 1] = RESOLUTION - 1;
74}

76OpenGLOscilloscope::~OpenGLOscilloscope() {
77 delete[] line_data_;
78 delete[] line_indices_;
79}

94void OpenGLOscilloscope::init(OpenGLContext& open_gl_context) {
95 gl_ready_ = false;
96 position_.reset();
97 open_gl_context.extensions.glGenBuffers(1, &line_buffer_);
98 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
99
100 GLsizeiptr vert_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(float));
101 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
102 line_data_, GL_STATIC_DRAW);
103
104 open_gl_context.extensions.glGenBuffers(1, &line_indices_buffer_);
105 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, line_indices_buffer_);
106
107 GLsizeiptr line_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(int));
108 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, line_size,
109 line_indices_, GL_STATIC_DRAW);
110
111 const char* vertex_shader = Shaders::getShader(Shaders::kOscilloscopeVertex);
112 const char* fragment_shader = Shaders::getShader(Shaders::kOscilloscopeFragment);
113
114 shader_ = std::make_unique<OpenGLShaderProgram>(open_gl_context);
115
116 if (shader_->addVertexShader(OpenGLHelpers::translateVertexShaderToV3(vertex_shader)) &&
117 shader_->addFragmentShader(OpenGLHelpers::translateFragmentShaderToV3(fragment_shader)) &&
118 shader_->link()) {
119 shader_->use();
120 position_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "position");
121 gl_ready_ = line_buffer_ != 0 && line_indices_buffer_ != 0 &&
122 position_->attributeID != static_cast<GLuint>(-1);
123 }
124 if (!gl_ready_)
125 Logger::writeToLog("[OpenGL] Oscilloscope initialization failed: " + shader_->getLastError());
126}

128void OpenGLOscilloscope::drawLines(OpenGLContext& open_gl_context) {
129 if (synth_ != nullptr) {
130 synth_->copyOutputMemory(output_snapshot_);
131 output_memory_ = output_snapshot_.data();
132 }
133 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
134 glEnable(GL_LINE_SMOOTH);
135 glHint(GL_LINE_SMOOTH_HINT, GL_NICEST);
136 glLineWidth(1.0f);
137
138 setViewPort(open_gl_context);
139
140 if (output_memory_) {
141 for (int i = 0; i < RESOLUTION; ++i) {
142 float memory_spot = (1.0f * i * mopo::MEMORY_RESOLUTION) / RESOLUTION;
143 int memory_index = memory_spot;
144 float remainder = memory_spot - memory_index;
145 line_data_[2 * i + 1] = mopo::utils::interpolate(output_memory_[memory_index],
146 output_memory_[memory_index + 1],
147 remainder);
148 }
149
150 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
151
152 GLsizeiptr vert_size = static_cast<GLsizeiptr>(2 * RESOLUTION * sizeof(float));
153 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
154 line_data_, GL_STATIC_DRAW);
155
156 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
157 }
158
159 shader_->use();
160 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, line_buffer_);
161 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, line_indices_buffer_);
162
163 open_gl_context.extensions.glVertexAttribPointer(position_->attributeID, 2, GL_FLOAT,
164 GL_FALSE, 2 * sizeof(float), 0);
165 open_gl_context.extensions.glEnableVertexAttribArray(position_->attributeID);
166 glDrawElements(GL_LINES, 2 * RESOLUTION, GL_UNSIGNED_INT, 0);
167
168 open_gl_context.extensions.glDisableVertexAttribArray(position_->attributeID);
169
170 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
171 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
172 glDisable(GL_LINE_SMOOTH);
173 MOPO_ASSERT(glGetError() == GL_NO_ERROR);
174}
