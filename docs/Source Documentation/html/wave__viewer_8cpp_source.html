58void WaveViewer::paint(juce::Graphics &g) {
59 g.drawImageWithin(background_,
60 0, 0, getWidth(), getHeight(), RectanglePlacement());
61
62 if (wave_phase_) {
63 if (phase_ >= 0.0 && phase_ < 1.0) {
64 float x = phaseToX(phase_);
65 g.setColour(Colour(Colors::Color_33ffffff));
66 g.fillRect(x - 0.5f, 0.0f, 1.0f, (float)getHeight());
67
68 float y = PADDING + (getHeight() - 2 * PADDING) * (1.0f - amp_) / 2.0f;
69
70 g.setColour(Colors::modulation);
71 g.fillEllipse(x - MARKER_WIDTH / 2.0f, y - MARKER_WIDTH / 2.0f,
72 MARKER_WIDTH, MARKER_WIDTH);
73 g.setColour(Colour(Colors::Color_ff000000));
74 g.fillEllipse(x - MARKER_WIDTH / 4.0f, y - MARKER_WIDTH / 4.0f,
75 MARKER_WIDTH / 2.0f, MARKER_WIDTH / 2.0f);
76 }
77 }
78}

80void WaveViewer::paintBackground(Graphics& g) {
81 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
82
83 g.fillAll(Colour(Colors::Color_ff424242));
84
85 g.setColour(Colour(Colors::Color_ff4a4a4a));
86 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
87 g.drawLine(x, 0, x, getHeight());
88 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
89 g.drawLine(0, y, getWidth(), y);
90
91 shadow.drawForPath(g, wave_path_);
92
93 g.setColour(Colors::graph_fill);
94 g.fillPath(wave_path_);
95
96 if (is_control_rate_)
97 g.setColour(Colors::modulation);
98 else
99 g.setColour(Colors::audio);
100
101 float line_width = 1.0f * getRatio();
102 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
103 g.strokePath(wave_path_, stroke);
104}

128void WaveViewer::timerCallback() {
129 if (phase_stretch_slider_ && isShowing()) {
130 auto* interface = findParentComponentOfClass<FullInterface>();
131 if (interface && interface->isAnimating()) {
132 const std::string name = phase_stretch_slider_->getName().toStdString();
133 if (interface->isDestinationModulated(name)) {
134 float value = 0.5f;
135 if (interface->copyModulatedValue(name, value) && std::isfinite(value)) {
136 modulated_phase_stretch_ = mopo::utils::clamp(value, 0.01f, 0.99f);
137 has_modulated_phase_stretch_ = true;
138 }
139 }
140 else
141 has_modulated_phase_stretch_ = false;
142
143 if (std::abs(getPhaseStretch() - displayed_phase_stretch_) >= 0.0001f)
144 resetWavePath();
145 }
146 }
147
148 if (wave_phase_) {
149 float phase = wave_phase_->buffer[0];
150 amp_ = wave_amp_->buffer[0];
151 if (phase != phase_) {
152 float last_x = phaseToX(phase_);
153 float new_x = phaseToX(phase);
154 phase_ = phase;
155 repaint(last_x - MARKER_WIDTH / 2.0f - 1, 0.0, MARKER_WIDTH + 2, getHeight());
156 repaint(new_x - MARKER_WIDTH / 2.0f - 1, 0.0, MARKER_WIDTH + 2, getHeight());
157 }
158 }

187float WaveViewer::getPhaseStretch() const {
188 float value = has_modulated_phase_stretch_ ? modulated_phase_stretch_
189 : phase_stretch_slider_ ? static_cast<float>(phase_stretch_slider_->getValue()) : 0.5f;
190 return std::isfinite(value) ? mopo::utils::clamp(value, 0.01f, 0.99f) : 0.5f;
191}

193void WaveViewer::drawRandom() {
194 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
195 float draw_width = getWidth();
196 float padding = getRatio() * PADDING;
197 float draw_height = getHeight() - 2.0f * padding;
198
199 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
200 for (int i = 0; i < NOISE_RESOLUTION; ++i) {
201 float t1 = (1.0f * i) / NOISE_RESOLUTION;
202 float t2 = (1.0f + i) / NOISE_RESOLUTION;
203 float val = amplitude * random_values[i];
204 wave_path_.lineTo(t1 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
205 wave_path_.lineTo(t2 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
206 }
207
208 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
209}

211void WaveViewer::drawSmoothRandom() {
212 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
213 float phase_stretch = getPhaseStretch();
214 float draw_width = getWidth();
215 float padding = getRatio() * PADDING;
216 float draw_height = getHeight() - 2.0f * padding;
217
218 float start_val = amplitude * random_values[0];
219 wave_path_.startNewSubPath(-50, getHeight() / 2.0f);
220 wave_path_.lineTo(0, padding + draw_height * ((1.0f - start_val) / 2.0f));
221 for (int i = 1; i < resolution_ - 1; ++i) {
222 float t = (1.0f * i) / resolution_;
223 float phase = mopo::utils::remapPhase(t, phase_stretch) * (NOISE_RESOLUTION - 1);
224 int index = (int)phase;
225 phase = mopo::PI * (phase - index);
226 float val = amplitude * mopo::utils::interpolate(random_values[index],
227 random_values[index + 1],
228 0.5f - cosf(phase) / 2.0f);
229 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
230 }
231
232 float end_val = amplitude * random_values[NOISE_RESOLUTION - 1];
233 wave_path_.lineTo(getWidth(), padding + draw_height * ((1.0f - end_val) / 2.0f));
234 wave_path_.lineTo(getWidth() + 50, getHeight() / 2.0f);
235
236}

238void WaveViewer::resetWavePath() {
239 if (!background_.isValid())
240 return;
241
242 wave_path_.clear();
243
244 if (wave_slider_ == nullptr)
245 return;
246
247 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
248 float phase_stretch = getPhaseStretch();
249 displayed_phase_stretch_ = phase_stretch;
250 float draw_width = getWidth();
251 float padding = getRatio() * PADDING;
252 float draw_height = getHeight() - 2.0f * padding;
253
254 mopo::Wave::Type type = static_cast<mopo::Wave::Type>(static_cast<int>(wave_slider_->getValue()));
255
256 if (type < mopo::Wave::Type::WhiteNoise) {
257 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
258 for (int i = 1; i < resolution_ - 1; ++i) {
259 float t = (1.0f * i) / resolution_;
260 float val = amplitude * mopo::Wave::wave(type, mopo::utils::remapPhase(t, phase_stretch));
261 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
262 }
263
264 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
265 }
266 else if (type == mopo::Wave::Type::WhiteNoise)
267 drawRandom();
268 else
269 drawSmoothRandom();
270
271 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
272 jassert(display != nullptr);
273
274 float scale = display->scale;
275 Graphics g(background_);
276 g.addTransform(AffineTransform::scale(scale, scale));
277 paintBackground(g);
278
279 repaint();
280}
