70void OpenGLWaveViewer::paintBackground() {
71 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
72
73 if (getWidth() <= 0 || getHeight() <= 0)
74 return;
75
76 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
77 jassert(display != nullptr);
78
79 float scale = display->scale;
80 background_image_ = Image(Image::ARGB, scale * getWidth(), scale * getHeight(), true);
81 {
82 Graphics g(background_image_);
83 g.addTransform(AffineTransform::scale(scale, scale));
84 // Fond de la grille d'affichage de la courbe d'onde
85 g.fillAll(Colour(Colors::Color_ff424242));
86 // Quadrillage de la grille d'affichage de la courbe d'onde
87 g.setColour(Colour(Colors::Color_ff4a4a4a));
88 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
89 g.drawLine(x, 0, x, getHeight());
90 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
91 g.drawLine(0, y, getWidth(), y);
92
93 shadow.drawForPath(g, wave_path_);
94
95 g.setColour(Colors::graph_fill);
96 g.fillPath(wave_path_);
97
98 g.setColour(Colors::modulation);
99 float line_width = 1.0f * getHeight() / 75.0f;
100 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
101 g.strokePath(wave_path_, stroke);
102 }
103
104 background_.updateBackgroundImage(background_image_);
105}

107void OpenGLWaveViewer::paintPositionImage() {
108 int min_image_width = roundToInt(2 * MARKER_WIDTH);
109 int image_width = mopo::utils::nextPowerOfTwo(min_image_width);
110 int marker_width = MARKER_WIDTH;
111 int image_height = roundToInt(2 * IMAGE_HEIGHT);
112 position_image_ = Image(Image::ARGB, image_width, image_height, true);
113 Graphics g(position_image_);
114
115 g.setColour(Colour(Colors::Color_77ffffff));
116 g.fillRect(image_width / 2.0f - 0.5f, 0.0f, 1.0f, 1.0f * image_height);
117
118 g.setColour(Colors::modulation);
119 g.fillEllipse(image_width / 2 - marker_width / 2, image_height / 2 - marker_width / 2,
120 marker_width, marker_width);
121 g.setColour(Colour(Colors::Color_ff000000));
122 g.fillEllipse(image_width / 2 - marker_width / 4, image_height / 2 - marker_width / 4,
123 marker_width / 2, marker_width / 2);
124}

175float OpenGLWaveViewer::getPhaseStretch() const {
176 float value = has_modulated_phase_stretch_ ? modulated_phase_stretch_
177 : phase_stretch_slider_ ? static_cast<float>(phase_stretch_slider_->getValue()) : 0.5f;
178 return std::isfinite(value) ? mopo::utils::clamp(value, 0.01f, 0.99f) : 0.5f;
179}

181void OpenGLWaveViewer::timerCallback() {
182 if (!phase_stretch_slider_ || !isShowing())
183 return;
184 auto* interface = findParentComponentOfClass<FullInterface>();
185 if (!interface || !interface->isAnimating())
186 return;
187
188 const std::string name = phase_stretch_slider_->getName().toStdString();
189 if (interface->isDestinationModulated(name)) {
190 float value = 0.5f;
191 if (interface->copyModulatedValue(name, value) && std::isfinite(value)) {
192 modulated_phase_stretch_ = mopo::utils::clamp(value, 0.01f, 0.99f);
193 has_modulated_phase_stretch_ = true;
194 }
195 }
196 else
197 has_modulated_phase_stretch_ = false;
198
199 if (std::abs(getPhaseStretch() - displayed_phase_stretch_) >= 0.0001f)
200 resetWavePath();
201}

203void OpenGLWaveViewer::drawRandom() {
204 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
205 float draw_width = getWidth();
206 float padding = getRatio() * PADDING;
207 float draw_height = getHeight() - 2.0f * padding;
208
209 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
210 for (int i = 0; i < NOISE_RESOLUTION; ++i) {
211 float t1 = (1.0f * i) / NOISE_RESOLUTION;
212 float t2 = (1.0f + i) / NOISE_RESOLUTION;
213 float val = amplitude * random_values[i];
214 wave_path_.lineTo(t1 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
215 wave_path_.lineTo(t2 * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
216 }
217
218 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
219}

221void OpenGLWaveViewer::drawSmoothRandom() {
222 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
223 float phase_stretch = getPhaseStretch();
224 float draw_width = getWidth();
225 float padding = getRatio() * PADDING;
226 float draw_height = getHeight() - 2.0f * padding;
227
228 float start_val = amplitude * random_values[0];
229 wave_path_.startNewSubPath(-50, getHeight() / 2.0f);
230 wave_path_.lineTo(0, PADDING + draw_height * ((1.0f - start_val) / 2.0f));
231 for (int i = 1; i < resolution_ - 1; ++i) {
232 float t = (1.0f * i) / resolution_;
233 float phase = mopo::utils::remapPhase(t, phase_stretch) * (NOISE_RESOLUTION - 1);
234 int index = (int)phase;
235 phase = mopo::PI * (phase - index);
236 float val = amplitude * mopo::utils::interpolate(random_values[index],
237 random_values[index + 1],
238 0.5f - cosf(phase) / 2.0f);
239 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
240 }
241
242 float end_val = amplitude * random_values[NOISE_RESOLUTION - 1];
243 wave_path_.lineTo(getWidth(), padding + draw_height * ((1.0f - end_val) / 2.0f));
244 wave_path_.lineTo(getWidth() + 50, getHeight() / 2.0f);
245
246}

248void OpenGLWaveViewer::resetWavePath() {
249 wave_path_.clear();
250
251 if (wave_slider_ == nullptr) {
252 publishRenderGeometry();
253 return;
254 }
255
256 float amplitude = amplitude_slider_ ? amplitude_slider_->getValue() : 1.0f;
257 float phase_stretch = getPhaseStretch();
258 displayed_phase_stretch_ = phase_stretch;
259 float draw_width = getWidth();
260 float padding = getRatio() * PADDING;
261 float draw_height = getHeight() - 2.0f * padding;
262
263 mopo::Wave::Type type = static_cast<mopo::Wave::Type>(static_cast<int>(wave_slider_->getValue()));
264
265 if (type < mopo::Wave::Type::WhiteNoise) {
266 wave_path_.startNewSubPath(0, getHeight() / 2.0f);
267 for (int i = 1; i < resolution_ - 1; ++i) {
268 float t = (1.0f * i) / resolution_;
269 float val = amplitude * mopo::Wave::wave(type, mopo::utils::remapPhase(t, phase_stretch));
270 wave_path_.lineTo(t * draw_width, padding + draw_height * ((1.0f - val) / 2.0f));
271 }
272
273 wave_path_.lineTo(getWidth(), getHeight() / 2.0f);
274 }
275 else if (type == mopo::Wave::Type::WhiteNoise)
276 drawRandom();
277 else
278 drawSmoothRandom();
279
280 paintBackground();
281}

299void OpenGLWaveViewer::init(OpenGLContext& open_gl_context) {
300 paintPositionImage();
301
302 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
303 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
304

306 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
307 position_vertices_, GL_STATIC_DRAW);
308

310 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);

314 position_triangles_, GL_STATIC_DRAW);
315
316 background_.init(open_gl_context);

319void OpenGLWaveViewer::drawPosition(OpenGLContext& open_gl_context) {
320 open_gl_context.extensions.glActiveTexture(GL_TEXTURE0);
321 if (position_texture_.getWidth() != position_image_.getWidth())
322 position_texture_.loadImage(position_image_);
323
324 auto geometry = render_geometry_.load(std::memory_order_acquire);
325 if (synth_ == nullptr || !geometry || geometry->width <= 0.0f || geometry->height <= 0.0f ||
326 telemetry_indices_[0] < 0 || telemetry_indices_[1] < 0)
327 return;
328
329 (void)synth_->copyVisualTelemetry(telemetry_indices_, telemetry_values_);
330 if (telemetry_values_[0] <= 0.0f)
331 return;
332
333 float x = 2.0f * telemetry_values_[0] - 1.0f;
334 float y = (geometry->height - 2 * geometry->padding) * telemetry_values_[1] / geometry->height;
335
336 glEnable(GL_BLEND);
337 glBlendFunc(GL_ONE, GL_ONE_MINUS_SRC_ALPHA);
338
339 int draw_width = static_cast<int>(geometry->width);
340 int draw_height = static_cast<int>(geometry->height);
341
342 float ratio = geometry->height / 75.0f;
343
344 float position_height = ratio * (0.5f * position_texture_.getHeight()) / draw_height;
345 float position_width = ratio * (0.5f * position_texture_.getWidth()) / draw_width;
346 position_vertices_[0] = x - position_width;
347 position_vertices_[1] = y + position_height;
348 position_vertices_[4] = x - position_width;
349 position_vertices_[5] = y - position_height;
350 position_vertices_[8] = x + position_width;
351 position_vertices_[9] = y - position_height;
352 position_vertices_[12] = x + position_width;
353 position_vertices_[13] = y + position_height;
354
355 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
356 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
357 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
358 position_vertices_, GL_STATIC_DRAW);
359
360 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
361 position_texture_.bind();
362 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
363 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
364 background_.shader()->use();
365
366 if (background_.texture_uniform() != nullptr)
367 background_.texture_uniform()->set(0);
368
369 background_.enableAttributes(open_gl_context);
370 glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);
371 background_.disableAttributes(open_gl_context);
372
373 position_texture_.unbind();
374
375 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
376 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
377}
