95 void Delay::process() {
96
97 MOPO_ASSERT(inputMatchesBufferSize(static_cast<int>(Inputs::Audio)));
98
99 auto audio = std::span<const mopo_float>(input(static_cast<int>(Inputs::Audio))->source->buffer, buffer_size_);
100 const Output* right_source = input(static_cast<int>(Inputs::AudioRight))->source;
101 if (right_source == &null_source_)
102 right_source = input(static_cast<int>(Inputs::Audio))->source;
103 MOPO_ASSERT(right_source->buffer_size >= buffer_size_);
104 const mopo_float* right_audio = right_source->buffer;
105 mopo_float* right_dest = output(1)->buffer;
106 auto dest = std::span<mopo_float>(output()->buffer, buffer_size_);
107
108 // Read target wet value and compute a square-root curve for perceptual
109 // loudness compensation. We also compute the corresponding dry value.
110 mopo_float wet = utils::clamp(input(static_cast<int>(Inputs::Wet))->at(0), 0.0, 1.0);
111 mopo_float new_wet = sqrt(wet);
112 mopo_float new_dry = sqrt(1.0 - wet);
113 // Compute per-sample increments to smoothly interpolate parameter values
114 // across the current buffer to avoid clicks.
115 mopo_float wet_inc = (new_wet - current_wet_) / buffer_size_;
116 mopo_float dry_inc = (new_dry - current_dry_) / buffer_size_;
117
118 // Feedback parameter interpolation
119 mopo_float new_feedback = input(static_cast<int>(Inputs::Feedback))->at(0);
120 mopo_float feedback_inc = (new_feedback - current_feedback_) / buffer_size_;
121
122 // Delay period (in samples) clamped to [2, memory_size - 1]
123 mopo_float new_period = utils::clamp(input(static_cast<int>(Inputs::SampleDelay))->at(0), 2.0, memory_->getSize() - 1.0);
124 mopo_float period_inc = (new_period - current_period_) / buffer_size_;
125
126 // Process each sample in the block, updating smoothed parameters and
127 // calling tick() which performs the per-sample memory read/write.
128 std::ranges::for_each(std::views::iota(0, buffer_size_), [&](int i) {
129 current_feedback_ += feedback_inc;
130 current_wet_ += wet_inc;
131 current_dry_ += dry_inc;
132 current_period_ += period_inc;
133 tick(i, audio.data(), dest.data());
134 const mopo_float right_read = right_memory_->get(current_period_);
135 right_memory_->push(right_audio[i] + right_read * current_feedback_);
136 right_dest[i] = current_dry_ * right_audio[i] + current_wet_ * right_read;
137 MOPO_ASSERT(std::isfinite(right_dest[i]));
138 });
139 }

150 inline void Delay::tick(int i, const mopo_float* audio, mopo_float* dest) {
151 mopo_float read = memory_->get(current_period_);
152 memory_->push(audio[i] + read * current_feedback_);
153 dest[i] = current_dry_ * audio[i] + current_wet_ * read;
154 MOPO_ASSERT(std::isfinite(dest[i]));
155 }
