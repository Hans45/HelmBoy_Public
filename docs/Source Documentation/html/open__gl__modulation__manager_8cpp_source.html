42#define INDICES_PER_METER 6
43#define POINTS_PER_METER 4

45OpenGLModulationManager::OpenGLModulationManager(
46 SynthBase* synth,
47 mopo::output_map modulation_sources,
48 std::map<std::string, ModulationButton*> modulation_buttons,
49 std::map<std::string, SynthSlider*> sliders,
50 std::map<std::string, SynthButton*> buttons,
51 mopo::output_map mono_modulations,
52 mopo::output_map poly_modulations) {
53 static const int quad_triangles[6] {
54 0, 1, 2,
55 2, 3, 0
56 };
57
58 modulation_buttons_ = modulation_buttons;
59 synth_ = synth;
60 modulation_sources_ = modulation_sources;
61 setInterceptsMouseClicks(false, true);
62
63 current_modulator_ = "";
64
65 polyphonic_destinations_ = std::make_unique<Component>();
66 polyphonic_destinations_->setInterceptsMouseClicks(false, true);
67
68 monophonic_destinations_ = std::make_unique<Component>();
69 monophonic_destinations_->setInterceptsMouseClicks(false, true);
70
71 for (auto& mod_button : modulation_buttons_) {
72 mod_button.second->addListener(this);
73 mod_button.second->addDisconnectListener(this);
74
75 // Create modulation highlight overlays.
76 std::string name = mod_button.second->getName().toStdString();
77 ModulationHighlight* overlay = new ModulationHighlight();
78 addChildComponent(overlay);
79 overlay_lookup_[name] = overlay;
80 overlay->setName(name);
81 Rectangle<int> local_bounds = mod_button.second->getBoundsInParent();
82 overlay->setBounds(mod_button.second->getParentComponent()->localAreaToGlobal(local_bounds));
83 }
84
85 slider_model_lookup_ = sliders;
86 button_model_lookup_ = buttons;
87 vertices_ = new float[FLOATS_PER_METER * slider_model_lookup_.size()];
88 triangles_ = new int[INDICES_PER_METER * slider_model_lookup_.size()];
89
90 int i = 0;
91 for (auto& slider : slider_model_lookup_) {
92 std::string name = slider.first;
93 const mopo::Output* mono_total = mono_modulations[name];
94 const mopo::Output* poly_total = poly_modulations[name];
95
96 float* meter_vertices = vertices_ + (i * FLOATS_PER_METER);
97 memset(meter_vertices, 0, FLOATS_PER_METER * sizeof(float));
98
99 int* meter_triangles = triangles_ + (i * INDICES_PER_METER);
100 for (int t = 0; t < INDICES_PER_METER; ++t)
101 meter_triangles[t] = i * POINTS_PER_METER + quad_triangles[t];
102
103 slider.second->addSliderListener(this);
104
105 // Create modulation meter.
106 if (mono_total) {
107 std::string name = slider.second->getName().toStdString();
108 OpenGLModulationMeter* meter = new OpenGLModulationMeter(synth_, mono_total, poly_total,
109 slider.second, std::span<float>(meter_vertices, FLOATS_PER_METER));
110 addChildComponent(meter);
111 meter_lookup_[name] = meter;
112 meter->setName(name);
113 Rectangle<int> local_bounds = slider.second->getBoundsInParent();
114 meter->setBounds(slider.second->getParentComponent()->localAreaToGlobal(local_bounds));
115 }
116
117 // Create modulation slider.
118 ModulationSlider* mod_slider = new ModulationSlider(slider.second);
119 mod_slider->setLookAndFeel(ModulationLookAndFeel::instance());
120 mod_slider->addListener(this);
121 if (poly_total)
122 polyphonic_destinations_->addAndMakeVisible(mod_slider);
123 else
124 monophonic_destinations_->addAndMakeVisible(mod_slider);
125
126 slider_lookup_[name] = mod_slider;
127 owned_sliders_.push_back(mod_slider);
128
129 ++i;
130 }
131
132 for (auto& button : button_model_lookup_) {
133 const mopo::Output* poly_total = poly_modulations[button.first];
134
135 button.second->addButtonListener(this);
136
137 ButtonModulationSlider* mod_slider = new ButtonModulationSlider(button.second);
138 mod_slider->setLookAndFeel(ModulationLookAndFeel::instance());
139 mod_slider->addListener(this);
140 if (poly_total)
141 polyphonic_destinations_->addAndMakeVisible(mod_slider);
142 else
143 monophonic_destinations_->addAndMakeVisible(mod_slider);
144
145 slider_lookup_[button.first] = mod_slider;
146 owned_sliders_.push_back(mod_slider);
147 }
148
149 addAndMakeVisible(polyphonic_destinations_.get());
150 addAndMakeVisible(monophonic_destinations_.get());
151
152 forgetModulator();
153}

157 button->removeListener(this);

179void OpenGLModulationManager::resized() {
180 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
181 polyphonic_destinations_->setBounds(getBounds());
182 monophonic_destinations_->setBounds(getBounds());
183
184 // Update modulation slider locations.
185 for (auto& slider : slider_lookup_) {
186 setDestinationSliderBounds(slider.first, slider.second);
187 }
188
189 // Update modulation meter locations.
190 for (auto& meter : meter_lookup_) {
191 Slider* model = slider_model_lookup_[meter.first];
192 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
193 meter.second->setBounds(local_top_left.x, local_top_left.y,
194 model->getWidth(), model->getHeight());
195 if (parent) {
196 int num_modulations = parent->getSynth()->getNumModulations(meter.first);
197 meter.second->setModulated(num_modulations);
198 meter.second->setVisible(num_modulations && model->isVisible());
199 }
200 }
201
202 // Update modulation highlight overlay locations.
203 for (auto& overlay : overlay_lookup_) {
204 ModulationButton* model = modulation_buttons_[overlay.first];
205 Point<float> local_top_left = getLocalPoint(model, Point<float>(0.0f, 0.0f));
206 overlay.second->setBounds(local_top_left.x, local_top_left.y,
207 model->getWidth(), model->getHeight());
208 }
209
210 OpenGLComponent::resized();

257void OpenGLModulationManager::init(OpenGLContext& open_gl_context) {
258 gl_ready_ = false;
259 position_.reset();
260 coordinates_.reset();
261 range_.reset();
262 radius_uniform_.reset();
263 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
264 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
265
266 int num_meters = slider_model_lookup_.size();
267 GLsizeiptr vert_size = static_cast<GLsizeiptr>(num_meters * FLOATS_PER_METER * sizeof(float));
268 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
269 vertices_, GL_STATIC_DRAW);
270
271 open_gl_context.extensions.glGenBuffers(1, &triangle_buffer_);
272 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
273
274 GLsizeiptr tri_size = static_cast<GLsizeiptr>(num_meters * INDICES_PER_METER * sizeof(int));
275 open_gl_context.extensions.glBufferData(GL_ELEMENT_ARRAY_BUFFER, tri_size,
276 triangles_, GL_STATIC_DRAW);
277
278 const char* vertex_shader = Shaders::getShader(Shaders::kModulationVertex);
279 const char* fragment_shader = Shaders::getShader(Shaders::kModulationFragment);
280
281 shader_ = std::make_unique<OpenGLShaderProgram>(open_gl_context);
282
283 if (shader_->addVertexShader(OpenGLHelpers::translateVertexShaderToV3(vertex_shader)) &&
284 shader_->addFragmentShader(OpenGLHelpers::translateFragmentShaderToV3(fragment_shader)) &&
285 shader_->link()) {
286 shader_->use();
287 position_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "position");
288 coordinates_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "coordinates");
289 range_ = std::make_unique<OpenGLShaderProgram::Attribute>(*shader_, "range");
290 radius_uniform_ = std::make_unique<OpenGLShaderProgram::Uniform>(*shader_, "radius");
291 gl_ready_ = vertex_buffer_ != 0 && triangle_buffer_ != 0 &&
292 position_->attributeID != static_cast<GLuint>(-1) &&
293 coordinates_->attributeID != static_cast<GLuint>(-1) &&
294 range_->attributeID != static_cast<GLuint>(-1) && radius_uniform_->uniformID >= 0;
295 }
296 if (!gl_ready_)
297 Logger::writeToLog("[OpenGL] Modulation manager initialization failed: " + shader_->getLastError());
298}

300void OpenGLModulationManager::render(OpenGLContext& open_gl_context, bool animate) {
301 if (!gl_ready_ || !animate)
302 return;
303
304 for (auto& meter : meter_lookup_) {
305 meter.second->updateDrawing();
306 }
307
308 glEnable(GL_BLEND);
309 glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
310 setViewPort(open_gl_context);
311
312 shader_->use();
313 radius_uniform_->set(0.9f);
314
315 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
316
317 int num_meters = slider_model_lookup_.size();
318 GLsizeiptr vert_size = static_cast<GLsizeiptr>(num_meters * FLOATS_PER_METER * sizeof(float));
319 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
320 vertices_, GL_STATIC_DRAW);
321 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
322
323 if (position_ != nullptr) {
324 open_gl_context.extensions.glVertexAttribPointer(position_->attributeID, 2, GL_FLOAT,
325 GL_FALSE, 6 * sizeof(float), 0);
326 open_gl_context.extensions.glEnableVertexAttribArray(position_->attributeID);
327 }
328
329 if (coordinates_ != nullptr) {
330 open_gl_context.extensions.glVertexAttribPointer(coordinates_->attributeID, 2, GL_FLOAT,
331 GL_FALSE, 6 * sizeof(float),
332 (GLvoid*)(2 * sizeof(float)));
333 open_gl_context.extensions.glEnableVertexAttribArray(coordinates_->attributeID);
334 }
335
336 if (range_ != nullptr) {
337 open_gl_context.extensions.glVertexAttribPointer(range_->attributeID, 2, GL_FLOAT,
338 GL_FALSE, 6 * sizeof(float),
339 (GLvoid*)(4 * sizeof(float)));
340 open_gl_context.extensions.glEnableVertexAttribArray(range_->attributeID);
341 }
342
343 glDrawElements(GL_TRIANGLES, num_meters * INDICES_PER_METER, GL_UNSIGNED_INT, 0);
344
345 if (position_ != nullptr)
346 open_gl_context.extensions.glDisableVertexAttribArray(position_->attributeID);
347
348 if (coordinates_ != nullptr)
349 open_gl_context.extensions.glDisableVertexAttribArray(coordinates_->attributeID);
350
351 if (range_ != nullptr)
352 open_gl_context.extensions.glDisableVertexAttribArray(range_->attributeID);
353
354 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
355 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
356}

390void OpenGLModulationManager::modulationsChanged(const std::string& destination) {
391 SynthGuiInterface* parent = findParentComponentOfClass<SynthGuiInterface>();
392 if (parent == nullptr)
393 return;
394
395 SynthBase* synth = parent->getSynth();
396 int num_modulations = synth->getNumModulations(destination);
397
398 float modulation_amount = 0.0f;
399 for (auto* connection : synth->getDestinationConnections(destination)) {
400 if (connection->source == current_modulator_) {
401 modulation_amount = connection->amount.value();
402 break;
403 }
404 }
405 auto modulation_slider = slider_lookup_.find(destination);
406 if (modulation_slider != slider_lookup_.end()) {
407 modulation_slider->second->setValue(modulation_amount, NotificationType::dontSendNotification);
408 modulation_slider->second->repaint();
409 }
410
411 if (meter_lookup_.count(destination)) {
412 meter_lookup_[destination]->setModulated(num_modulations);
413 meter_lookup_[destination]->setVisible(num_modulations && slider_model_lookup_[destination]->isVisible());
414 }
415 if (button_model_lookup_.count(destination))
416 button_model_lookup_[destination]->setModulated(num_modulations);
417}

440void OpenGLModulationManager::reset(const SynthGuiStateSnapshot& snapshot) {
441 for (auto& meter : meter_lookup_) {
442 int num_modulations = static_cast<int>(std::count_if(
443 snapshot.modulations.begin(), snapshot.modulations.end(),
444 [&meter](const auto& connection) { return connection.destination == meter.first; }));
445 meter.second->setModulated(num_modulations);
446 meter.second->setVisible(num_modulations && slider_model_lookup_[meter.first]->isVisible());
447 }
448
449 for (auto& button : button_model_lookup_) {
450 int num_modulations = static_cast<int>(std::count_if(
451 snapshot.modulations.begin(), snapshot.modulations.end(),
452 [&button](const auto& connection) { return connection.destination == button.first; }));
453 button.second->setModulated(num_modulations);
454 }
455
456 for (auto& slider : slider_lookup_) {
457 float value = 0.0f;
458 const std::string destination = slider.second->getName().toStdString();
459 for (const auto& connection : snapshot.modulations) {
460 if (connection.source == current_modulator_ && connection.destination == destination) {
461 value = static_cast<float>(connection.amount);
462 break;
463 }
464 }
465 slider.second->setValue(value, NotificationType::dontSendNotification);
466 slider.second->repaint();
467 }
468}

517bool OpenGLModulationManager::modulationGesture(SynthButton* button, const ModifierKeys& mods) {
518 if (current_modulator_.empty())
519 return false;
520
521 std::string destination = button->getName().toStdString();
522 if (!button_model_lookup_.count(destination) || !slider_lookup_.count(destination))
523 return false;
524
525 Slider* slider = slider_lookup_[destination];
526 double threshold = mods.isShiftDown() ? -0.5 : 0.5;
527 if (std::abs(slider->getValue() - threshold) < 0.0001)
528 threshold = slider->getDoubleClickReturnValue();
529
530 slider->setValue(threshold);
531 return true;
532}
