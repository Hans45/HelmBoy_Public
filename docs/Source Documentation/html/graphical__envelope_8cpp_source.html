49void GraphicalEnvelope::paint(Graphics& g) {
50 const float line_width = 1.0f * getHeight() / 150.0f;
51 const PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
52 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
53
54 g.fillAll(Colour(Colors::Color_ff424242));
55
56 g.setColour(Colour(Colors::Color_ff4a4a4a));
57 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
58 g.drawLine(x, 0, x, getHeight());
59 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
60 g.drawLine(0, y, getWidth(), y);
61
62 shadow.drawForPath(g, envelope_line_);
63 g.setColour(Colors::graph_fill);
64 g.fillPath(envelope_line_);
65
66 g.setColour(Colour(Colors::Color_ff505050));
67 g.drawLine(getAttackX(), 0.0f, getAttackX(), getHeight());
68 g.drawLine(getDecayX(), getSustainY(), getDecayX(), getHeight());
69
70 g.setColour(Colors::modulation);
71 g.strokePath(envelope_line_, stroke);
72
73 float hover_line_x = -20;
74 if (attack_hover_)
75 hover_line_x = getAttackX();
76 else if (decay_hover_)
77 hover_line_x = getDecayX();
78 else if (release_hover_)
79 hover_line_x = getReleaseX();
80
81 g.setColour(Colour(Colors::Color_bbffffff));
82 g.fillRect(hover_line_x - 0.5f, 0.0f, 1.0f, 1.0f * getHeight());
83
84 if (sustain_hover_) {
85 if (mouse_down_) {
86 g.setColour(Colour(Colors::Color_11ffffff));
87 g.fillEllipse(getDecayX() - 20.0, getSustainY() - 20.0, 40.0, 40.0);
88 }
89
90 g.setColour(Colour(Colors::Color_bbffffff));
91 g.drawEllipse(getDecayX() - 7.0, getSustainY() - 7.0, 14.0, 14.0, 1.0);
92 }
93 else if (mouse_down_) {
94 g.setColour(Colour(Colors::Color_11ffffff));
95 g.fillRect(hover_line_x - 10.0f, 0.0f, 20.0f, 1.0f * getHeight());
96 }
97
98}

104void GraphicalEnvelope::mouseMove(const MouseEvent& e) {
105 float x = e.getPosition().x;
106 float attack_delta = fabs(x - getAttackX());
107 float decay_delta = fabs(x - getDecayX());
108 float release_delta = fabs(x - getReleaseX());
109 float sustain_delta = fabs(e.getPosition().y - getSustainY());
110
111 bool a_hover = attack_delta < decay_delta && attack_delta < HOVER_DISTANCE;
112 bool d_hover = !attack_hover_ && decay_delta < release_delta && decay_delta < HOVER_DISTANCE;
113 bool r_hover = !decay_hover_ && release_delta < HOVER_DISTANCE;
114 bool s_hover = !a_hover && !r_hover && x > getDecayX() - HOVER_DISTANCE &&
115 x < getDecayX() + HOVER_DISTANCE && sustain_delta < HOVER_DISTANCE;
116
117 if (a_hover != attack_hover_ || d_hover != decay_hover_ ||
118 s_hover != sustain_hover_ || r_hover != release_hover_) {
119 attack_hover_ = a_hover;
120 decay_hover_ = d_hover;
121 sustain_hover_ = s_hover;
122 release_hover_ = r_hover;
123 repaint();
124 }
125}

167float GraphicalEnvelope::getAttackX() {
168 if (!attack_slider_)
169 return 0.0;
170
171 double percent = attack_slider_->valueToProportionOfLength(attack_slider_->getValue());
172 return 1 + (getWidth() - 1) * percent * ATTACK_RANGE_PERCENT;
173}
174
175float GraphicalEnvelope::getDecayX() {
176 if (!decay_slider_)
177 return 0.0;
178
179 double percent = decay_slider_->valueToProportionOfLength(decay_slider_->getValue());
180 return getAttackX() + getWidth() * percent * DECAY_RANGE_PERCENT;
181}
182
183float GraphicalEnvelope::getSustainY() {
184 if (!sustain_slider_)
185 return 0.0;
186
187 double percent = sustain_slider_->valueToProportionOfLength(sustain_slider_->getValue());
188 return getHeight() * (1.0 - percent);
189}
190
191float GraphicalEnvelope::getReleaseX() {
192 if (!release_slider_)
193 return 0.0;
194
195 double percent = release_slider_->valueToProportionOfLength(release_slider_->getValue());
196 return getDecayX() + getWidth() * percent * (1.0 - DECAY_RANGE_PERCENT - ATTACK_RANGE_PERCENT);
197}

259void GraphicalEnvelope::resetEnvelopeLine() {
260 envelope_line_.clear();
261 envelope_line_.startNewSubPath(1, getHeight());
262 envelope_line_.lineTo(getAttackX(), 0.0f);
263 envelope_line_.quadraticTo(0.5f * (getAttackX() + getDecayX()), getSustainY(),
264 getDecayX(), getSustainY());
265
266 envelope_line_.quadraticTo(0.5f * (getReleaseX() + getDecayX()), getHeight(),
267 getReleaseX(), getHeight());
268}
