55 class WaveLookup {
56 public:
60 WaveLookup() {
61 // Sin lookup table.
62 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i)
63 sin_[i] = std::sin((2.0 * mopo::PI * i) / mopo::LOOKUP_SIZE);
64
65 // Square lookup table.
66 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
67 int p = i;
68 constexpr mopo_float scale = 4.0 / mopo::PI;
69 square_[0][i] = scale * sin_[p];
70 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
71 p = (p + i) % mopo::LOOKUP_SIZE;
72 square_[h][i] = square_[h - 1][i];
73 if (h % 2 == 0)
74 square_[h][i] += scale * sin_[p] / static_cast<mopo_float>(h + 1);
75 }
76 }
77
78 // Saw lookup table.
79 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
80 int index = (i + (mopo::LOOKUP_SIZE / 2)) % mopo::LOOKUP_SIZE;
81 int p = i;
82 constexpr mopo_float scale = 2.0 / mopo::PI;
83 saw_[0][index] = scale * sin_[p];
84 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
85 p = (p + i) % mopo::LOOKUP_SIZE;
86 mopo_float harmonic = scale * sin_[p] / static_cast<mopo_float>(h + 1);
87 if (h % 2 == 0)
88 saw_[h][index] = saw_[h - 1][index] + harmonic;
89 else
90 saw_[h][index] = saw_[h - 1][index] - harmonic;
91 }
92 }
93
94 // Triangle lookup table.
95 for (int i = 0; i < mopo::LOOKUP_SIZE + 1; ++i) {
96 int p = i;
97 constexpr mopo_float scale = 8.0 / (mopo::PI * mopo::PI);
98 triangle_[0][i] = scale * sin_[p];
99 for (int h = 1; h < mopo::MAX_HARMONICS; ++h) {
100 p = (p + i) % mopo::LOOKUP_SIZE;
101 triangle_[h][i] = triangle_[h - 1][i];
102 mopo_float harmonic = scale * sin_[p] / (static_cast<mopo_float>((h + 1) * (h + 1)));
103 if (h % 4 == 0)
104 triangle_[h][i] += harmonic;
105 else if (h % 2 == 0)
106 triangle_[h][i] -= harmonic;
107 }
108 }
109 }

172 template<size_t steps> requires (steps > 1)
173 [[nodiscard]] inline mopo_float step(mopo_float t, int harmonics) const {
174 constexpr mopo_float step_size = (1.0 * steps) / (steps - 1);
175 return step_size * (upsaw(t, harmonics) +
176 downsaw(steps * t, harmonics / steps) / steps);
177 }
178
182 template<size_t steps> requires (steps > 1)
183 [[nodiscard]] inline mopo_float pyramid(mopo_float t, int harmonics) const {
184 constexpr size_t squares = steps - 1;
185 constexpr mopo_float phase_increment = 1.0 / (2.0 * squares);
186
187 mopo_float phase = 0.75 + t;
188 mopo_float out = 0.0;
189
190 mopo_float integral;
191 for (size_t i = 0; i < squares; ++i) {
192 out += square(utils::mod(phase, &integral), harmonics);
193 phase += phase_increment;
194 }
195 out /= squares;
196 return out;
197 }

199 private:
200 // Make them 1 larger for wrapping.
201 mopo_float sin_[LOOKUP_SIZE + 1];
202 mopo_float square_[MAX_HARMONICS][LOOKUP_SIZE + 1];
203 mopo_float saw_[MAX_HARMONICS][LOOKUP_SIZE + 1];
204 mopo_float triangle_[MAX_HARMONICS][LOOKUP_SIZE + 1];
205 };

238 [[nodiscard]] static inline mopo_float blwave(Type waveform, mopo_float t,
239 mopo_float frequency) {
240 const WaveLookup* lookup = WaveLookup::instance();
241 if (std::fabs(frequency) < 1)
242 return Wave::wave(waveform, t);
243 int harmonics = mopo::HIGH_FREQUENCY / static_cast<int>(std::fabs(frequency)) - 1;
244 if (harmonics >= mopo::MAX_HARMONICS)
245 return Wave::wave(waveform, t);
246
247 switch (waveform) {
248 case Type::Sin:
249 return lookup->fullsin(t);
250 case Type::Triangle:
251 return lookup->triangle(t, harmonics);
252 case Type::Square:
253 return lookup->square(t, harmonics);
254 case Type::DownSaw:
255 return lookup->downsaw(t, harmonics);
256 case Type::UpSaw:
257 return lookup->upsaw(t, harmonics);
258 case Type::ThreeStep:
259 return lookup->step<3>(t, harmonics);
260 case Type::FourStep:
261 return lookup->step<4>(t, harmonics);
262 case Type::EightStep:
263 return lookup->step<8>(t, harmonics);
264 case Type::ThreePyramid:
265 return lookup->pyramid<3>(t, harmonics);
266 case Type::FivePyramid:
267 return lookup->pyramid<5>(t, harmonics);
268 case Type::NinePyramid:
269 return lookup->pyramid<9>(t, harmonics);
270 default:
271 return Wave::wave(waveform, t);
272 }
273 }
