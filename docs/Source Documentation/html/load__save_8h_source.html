50 static int compareElements(File a, File b) {
51 if (a.getFileName() == "Factory Presets")
52 return -1;
53 else if (b.getFileName() == "Factory Presets")
54 return 1;
55
56 if (a.getFileName() == "Old Factory Presets")
57 return 1;
58 else if (b.getFileName() == "Old Factory Presets")
59 return -1;
60
61 return a.getFullPathName().toLowerCase().compare(b.getFullPathName().toLowerCase());
62 }
