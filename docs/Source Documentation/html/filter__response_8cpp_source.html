114float FilterResponse::getPercentForMidiNote(float midi_note) {
115 float frequency = mopo::utils::midiNoteToFrequency(midi_note);
116
117 float response = 0.0f;
118 if (style_ == mopo::StateVariableFilter::Styles::Shelf)
119 response = fabs(filter_shelf_.getAmplitudeResponse(frequency));
120 else if (style_ == mopo::StateVariableFilter::Styles::Notch)
121 response = fabs(filter_notch_.getAmplitudeResponse(frequency));
122 else if (style_ == mopo::StateVariableFilter::Styles::Comb) {
123 const float phase = 2.0f * static_cast<float>(mopo::PI) * frequency * comb_period_seconds_;
124 const float denominator = std::sqrt(1.0f + comb_feedback_ * comb_feedback_ -
125 2.0f * comb_feedback_ * std::cos(phase));
126 response = 1.0f / denominator;
127 }
128 else {
129 float blend = filter_blend_slider_->getValue();
130 float low_pass_amount = mopo::utils::clamp(1.0 - blend, 0.0, 1.0);
131 float band_pass_amount = mopo::utils::clamp(1.0 - fabs(blend - 1.0), 0.0, 1.0);
132 float high_pass_amount = mopo::utils::clamp(blend - 1.0, 0.0, 1.0);
133
134 response = low_pass_amount * filter_low_.getAmplitudeResponse(frequency) +
135 band_pass_amount * filter_band_.getAmplitudeResponse(frequency) +
136 high_pass_amount * filter_high_.getAmplitudeResponse(frequency);
137
138 response = fabs(response);
139 if (style_ == mopo::StateVariableFilter::Styles::dB24)
140 response *= response;
141 }
142
143 float gain_db = mopo::utils::gainToDb(response);
144 return (gain_db - MIN_GAIN_DB) / (MAX_GAIN_DB - MIN_GAIN_DB);
145}

147void FilterResponse::resetResponsePath() {
148 static const int wrap_size = 10;
149
150 if (cutoff_slider_ == nullptr)
151 return;
152
153 filter_response_path_.clear();
154 filter_response_path_.startNewSubPath(-wrap_size, getHeight() + wrap_size);
155 float start_percent = getPercentForMidiNote(0.0);
156 float last_y = getHeight() * (1.0f - start_percent);
157 float last_slope = 0.0f;
158 float last_x = 0.0f;
159
160 filter_response_path_.lineTo(-wrap_size, last_y);
161
162 for (int i = 0; i < resolution_; ++i) {
163 float t = (1.0f * i) / (resolution_ - 1);
164 float midi_note = cutoff_slider_->proportionOfLengthToValue(t);
165 float percent = getPercentForMidiNote(midi_note);
166
167 float new_x = getWidth() * t;
168 float new_y = getHeight() * (1.0f - percent);
169 float new_slope = (new_y - last_y) / (new_x - last_x);
170 if (fabs(last_slope - new_slope) > DELTA_SLOPE_REDRAW_THRESHOLD ||
171 new_x - last_x > X_REDRAW_THRESHOLD) {
172 last_x = new_x;
173 last_y = new_y;
174 last_slope = new_slope;
175 filter_response_path_.lineTo(new_x, new_y);
176 }
177 }
178
179 float end_percent = getPercentForMidiNote(cutoff_slider_->getMaximum());
180
181 filter_response_path_.lineTo(getWidth() + wrap_size, getHeight() * (1.0f - end_percent));
182 filter_response_path_.lineTo(getWidth() + wrap_size, getHeight() + wrap_size);
183}

185void FilterResponse::computeFilterCoefficients() {
186 if (cutoff_slider_ == nullptr || resonance_slider_ == nullptr ||
187 filter_blend_slider_ == nullptr || filter_shelf_slider_ == nullptr) {
188 return;
189 }
190
191 mopo::StateVariableFilter::Shelves shelf = static_cast<mopo::StateVariableFilter::Shelves>(
192 static_cast<int>(filter_shelf_slider_->getValue()));
193 double frequency = mopo::utils::midiNoteToFrequency(cutoff_slider_->getValue());
194 double resonance = mopo::utils::magnitudeToQ(resonance_slider_->getValue());
195 double decibels = mopo::utils::interpolate(MIN_GAIN_DB, MAX_GAIN_DB,
196 resonance_slider_->getValue());
197 double gain = mopo::utils::dbToGain(decibels);
198
199 if (style_ == mopo::StateVariableFilter::Styles::dB24) {
200 resonance = sqrt(resonance);
201 gain = sqrt(gain);
202 }
203
204 const double cutoff_frequency = mopo::utils::midiNoteToFrequency(cutoff_slider_->getValue());
205 comb_period_seconds_ = static_cast<float>(mopo::utils::clamp(1.0 / cutoff_frequency,
206 0.00001, 0.1));
207 comb_feedback_ = static_cast<float>(mopo::utils::clamp(
208 0.95 * resonance_slider_->getValue(), 0.0, 0.95));
209
210 if (style_ == mopo::StateVariableFilter::Styles::Shelf) {
211 mopo::BiquadFilter::Type type = mopo::BiquadFilter::Type::LowShelf;
212 if (shelf == mopo::StateVariableFilter::Shelves::BandShelf)
213 type = mopo::BiquadFilter::Type::BandShelf;
214 else if (shelf == mopo::StateVariableFilter::Shelves::HighShelf)
215 type = mopo::BiquadFilter::Type::HighShelf;
216 filter_shelf_.computeCoefficients(type, frequency, 1.0, gain);
217 }
218 else if (style_ == mopo::StateVariableFilter::Styles::Notch) {
219 filter_notch_.computeCoefficients(mopo::BiquadFilter::Type::Notch,
220 frequency, resonance, 1.0);
221 }
222 else if (style_ != mopo::StateVariableFilter::Styles::Comb) {
223 filter_low_.computeCoefficients(mopo::BiquadFilter::Type::LowPass, frequency, resonance, 1.0);
224 filter_band_.computeCoefficients(mopo::BiquadFilter::Type::BandPass, frequency, resonance, 1.0);
225 filter_high_.computeCoefficients(mopo::BiquadFilter::Type::HighPass, frequency, resonance, 1.0);
226 }
227 resetResponsePath();
228}

230void FilterResponse::setFilterSettingsFromPosition(Point<int> position) {
231 if (cutoff_slider_) {
232 double percent = mopo::utils::clamp((1.0 * position.x) / getWidth(), 0.0, 1.0);
233 double frequency = cutoff_slider_->proportionOfLengthToValue(percent);
234 cutoff_slider_->setValue(frequency);
235 }
236 if (resonance_slider_) {
237 double percent = mopo::utils::clamp(1.0 - (1.0 * position.y) / getHeight(), 0.0, 1.0);
238 resonance_slider_->setValue(resonance_slider_->proportionOfLengthToValue(percent));
239 }
240
241 computeFilterCoefficients();
242}
