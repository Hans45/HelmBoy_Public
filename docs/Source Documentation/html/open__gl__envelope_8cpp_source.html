77void OpenGLEnvelope::paintBackground() {
78 static const DropShadow shadow(Colour(Colors::Color_bb000000), 5, Point<int>(0, 0));
79
80 if (getWidth() <= 0 || getHeight() <= 0)
81 return;
82
83 float ratio = getHeight() / 100.0f;
84
85 auto *display = Desktop::getInstance().getDisplays().getPrimaryDisplay();
86 jassert(display != nullptr);
87
88 float scale = display->scale;
89 background_image_ = Image(Image::ARGB, scale * getWidth(), scale * getHeight(), true);
90 {
91 Graphics g(background_image_);
92 g.addTransform(AffineTransform::scale(scale, scale));
93
94 g.fillAll(Colour(Colors::Color_ff424242));
95
96 g.setColour(Colour(Colors::Color_ff4a4a4a));
97 for (int x = 0; x < getWidth(); x += GRID_CELL_WIDTH)
98 g.drawLine(x, 0, x, getHeight());
99 for (int y = 0; y < getHeight(); y += GRID_CELL_WIDTH)
100 g.drawLine(0, y, getWidth(), y);
101
102 shadow.drawForPath(g, envelope_line_);
103 g.setColour(Colors::graph_fill);
104 g.fillPath(envelope_line_);
105
106 g.setColour(Colour(Colors::Color_ff505050));
107 if (delay_slider_)
108 g.drawLine(getDelayX(), 0.0f, getDelayX(), getHeight());
109 g.drawLine(getAttackX(), 0.0f, getAttackX(), getHeight());
110 g.drawLine(getHoldX(), 0.0f, getHoldX(), getHeight());
111 g.drawLine(getDecayX(), getSustainY(), getDecayX(), getHeight());
112
113 g.setColour(Colors::modulation);
114 float line_width = 1.0f * getHeight() / 150.0f;
115 PathStrokeType stroke(line_width, PathStrokeType::beveled, PathStrokeType::rounded);
116 g.strokePath(envelope_line_, stroke);
117
118 float hover_line_x = -20;
119 if (delay_hover_)
120 hover_line_x = getDelayX();
121 else if (attack_hover_)
122 hover_line_x = getAttackX();
123 else if (hold_hover_)
124 hover_line_x = getHoldX();
125 else if (decay_hover_)
126 hover_line_x = getDecayX();
127 else if (release_hover_)
128 hover_line_x = getReleaseX();
129
130 g.setColour(Colour(Colors::Color_bbffffff));
131 g.fillRect(hover_line_x - 0.5f, 0.0f, 1.0f, 1.0f * getHeight());
132
133 float grab_radius = 20.0f * ratio;
134 float hover_radius = 7.0f * ratio;
135 if (sustain_hover_) {
136 if (mouse_down_) {
137 g.setColour(Colour(Colors::Color_11ffffff));
138 g.fillEllipse(getDecayX() - grab_radius, getSustainY() - grab_radius,
139 2.0f * grab_radius, 2.0f * grab_radius);
140 }
141
142 g.setColour(Colour(Colors::Color_bbffffff));
143 g.drawEllipse(getDecayX() - hover_radius, getSustainY() - hover_radius,
144 2.0f * hover_radius, 2.0f * hover_radius, 1.0);
145 }
146 else if (mouse_down_) {
147 g.setColour(Colour(Colors::Color_11ffffff));
148 g.fillRect(hover_line_x - 10.0f, 0.0f, 20.0f, 1.0f * getHeight());
149 }
150
151 }
152
153 background_.updateBackgroundImage(background_image_);
154}

182void OpenGLEnvelope::mouseMove(const MouseEvent& e) {
183 float x = e.getPosition().x;
184 float y = e.getPosition().y;
185 float delay_delta = fabs(x - getDelayX());
186 float attack_delta = fabs(x - getAttackX());
187 float hold_delta = fabs(x - getHoldX());
188 float decay_delta = fabs(x - getDecayX());
189 float release_delta = fabs(x - getReleaseX());
190 float sustain_delta = fabs(y - getSustainY());
191
192 bool delay_is_hover = delay_slider_ && delay_delta < HOVER_DISTANCE &&
193 y > getHeight() - HOVER_DISTANCE;
194 bool hold_is_hover = hold_slider_ && hold_delta < HOVER_DISTANCE &&
195 y < HOVER_DISTANCE;
196 bool a_hover = !delay_is_hover && !hold_is_hover &&
197 attack_delta < decay_delta && attack_delta < HOVER_DISTANCE;
198 bool d_hover = !delay_is_hover && !hold_is_hover && !a_hover &&
199 decay_delta < release_delta && decay_delta < HOVER_DISTANCE;
200 bool r_hover = !delay_is_hover && !hold_is_hover && !a_hover && !d_hover &&
201 release_delta < HOVER_DISTANCE;
202 bool s_hover = !delay_is_hover && !hold_is_hover && !a_hover && !r_hover &&
203 x > getDecayX() - HOVER_DISTANCE &&
204 x < getDecayX() + HOVER_DISTANCE && sustain_delta < HOVER_DISTANCE;
205
206 if (delay_is_hover != delay_hover_ || a_hover != attack_hover_ ||
207 hold_is_hover != hold_hover_ || d_hover != decay_hover_ ||
208 s_hover != sustain_hover_ || r_hover != release_hover_) {
209 delay_hover_ = delay_is_hover;
210 attack_hover_ = a_hover;
211 hold_hover_ = hold_is_hover;
212 decay_hover_ = d_hover;
213 sustain_hover_ = s_hover;
214 release_hover_ = r_hover;
215 paintBackground();
216 }

314float OpenGLEnvelope::getReleaseRange() {
315 double delay_percent = delay_slider_ ?
316 delay_slider_->valueToProportionOfLength(delay_slider_->getValue()) : 0.0;
317 double hold_percent = hold_slider_ ?
318 hold_slider_->valueToProportionOfLength(hold_slider_->getValue()) : 0.0;
319 return 1.0f - ATTACK_RANGE_PERCENT - DECAY_RANGE_PERCENT -
320 delay_percent * DELAY_RANGE_PERCENT - hold_percent * HOLD_RANGE_PERCENT;
321}

408void OpenGLEnvelope::resetEnvelopeLine() {
409 envelope_line_.clear();
410 envelope_line_.startNewSubPath(1, getHeight());
411 envelope_line_.lineTo(getDelayX(), getHeight());
412 envelope_line_.lineTo(getAttackX(), 0.0f);
413 envelope_line_.lineTo(getHoldX(), 0.0f);
414 envelope_line_.quadraticTo(0.5f * (getHoldX() + getDecayX()), getSustainY(),
415 getDecayX(), getSustainY());
416
417 envelope_line_.quadraticTo(0.5f * (getReleaseX() + getDecayX()), getHeight(),
418 getReleaseX(), getHeight());
419
420 auto geometry = std::make_shared<GeometrySnapshot>();
421 geometry->line = envelope_line_;
422 geometry->width = static_cast<float>(getWidth());
423 geometry->height = static_cast<float>(getHeight());
424 geometry->delay_x = getDelayX();
425 geometry->attack_x = getAttackX();
426 geometry->hold_x = getHoldX();
427 geometry->decay_x = getDecayX();
428 geometry->release_x = getReleaseX();
429 geometry->sustain = sustain_slider_ != nullptr ? static_cast<float>(sustain_slider_->getValue()) : 0.0f;
430 geometry_snapshot_.store(std::move(geometry), std::memory_order_release);
431
432 paintBackground();
433}

435Point<float> OpenGLEnvelope::valuesToPosition(const GeometrySnapshot& geometry,
436 float phase, float amp,
437 float progress) {
438 float y = (1.0f - amp) * geometry.height;
439 float x = 0.0;
440
441 if (phase < 0.0 || phase > static_cast<float>(mopo::Envelope::State::Holding))
442 return Point<float>(-2.0, -2.0);
443
444 float delay_x = geometry.delay_x;
445 float attack_x = geometry.attack_x;
446 float hold_x = geometry.hold_x;
447 float decay_x = geometry.decay_x;
448 float release_x = geometry.release_x;
449 float sustain = geometry.sustain;
450
451 const int stage = static_cast<int>(phase);
452 if (stage == static_cast<int>(mopo::Envelope::State::Delaying)) {
453 x = 1.0f + progress * (delay_x - 1.0f);
454 }
455 else if (stage == static_cast<int>(mopo::Envelope::State::Attacking)) {
456 x = delay_x + amp * (attack_x - delay_x);
457 }
458 else if (stage == static_cast<int>(mopo::Envelope::State::Holding)) {
459 x = attack_x + progress * (hold_x - attack_x);
460 }
461 else if (stage == static_cast<int>(mopo::Envelope::State::Decaying)) {
462 float amp_diff = 1.0f - sustain;
463 if (amp_diff == 0.0f)
464 x = decay_x;
465 else {
466 float percent = (1.0f - amp) / amp_diff;
467 x = hold_x + percent * (decay_x - hold_x);
468 }
469 }
470 else {
471 float delta = release_x - decay_x;
472 x = decay_x + delta - delta * (amp / sustain);
473 }
474
475 Point<float> closest;
476 geometry.line.getNearestPoint(Point<float>(x, y), closest);
477 if (phase > 1.5f && phase < 2.5f && closest.x < decay_x) {
478 closest.x = decay_x;
479 closest.y = (1.0f - amp) * geometry.height;
480 }
481 if (phase > 0.5f && phase < 1.5f && closest.x > decay_x) {
482 closest.x = decay_x;
483 closest.y = (1.0f - amp) * geometry.height;
484 }
485 return Point<float>(2.0f * closest.x / geometry.width - 1.0f,
486 1.0f - 2.0f * closest.y / geometry.height);
487}

489void OpenGLEnvelope::init(OpenGLContext& open_gl_context) {
490 paintPositionImage();
491
492 open_gl_context.extensions.glGenBuffers(1, &vertex_buffer_);
493 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
494
495 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
496 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
497 position_vertices_, GL_STATIC_DRAW);
498
499 open_gl_context.extensions.glGenBuffers(1, &triangle_buffer_);
500 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);

506 background_.init(open_gl_context);

509void OpenGLEnvelope::drawPosition(OpenGLContext& open_gl_context) {
510 open_gl_context.extensions.glActiveTexture(GL_TEXTURE0);
511 if (position_texture_.getWidth() != position_image_.getWidth())
512 position_texture_.loadImage(position_image_);
513
514 auto geometry = geometry_snapshot_.load(std::memory_order_acquire);
515 if (synth_ == nullptr || !geometry || geometry->width <= 0.0f || geometry->height <= 0.0f ||
516 telemetry_indices_[0] < 0 || telemetry_indices_[1] < 0 || telemetry_indices_[2] < 0)
517 return;
518
519 (void)synth_->copyVisualTelemetry(telemetry_indices_, telemetry_values_);
520 if (telemetry_values_[1] <= 0.0f &&
521 static_cast<int>(telemetry_values_[0]) !=
522 static_cast<int>(mopo::Envelope::State::Delaying))
523 return;
524
525 Point<float> point = valuesToPosition(*geometry, telemetry_values_[0],
526 telemetry_values_[1], telemetry_values_[2]);
527 float x = point.x;
528 float y = point.y;
529
530 glEnable(GL_BLEND);
531 glBlendFunc(GL_ONE, GL_ONE_MINUS_SRC_ALPHA);
532
533 int draw_width = static_cast<int>(geometry->width);
534 int draw_height = static_cast<int>(geometry->height);
535
536 float ratio = geometry->height / 100.0f;
537
538 float position_height = ratio * (0.5f * position_texture_.getHeight()) / draw_height;
539 float position_width = ratio * (0.5f * position_texture_.getWidth()) / draw_width;
540 position_vertices_[0] = x - position_width;
541 position_vertices_[1] = y + position_height;
542 position_vertices_[4] = x - position_width;
543 position_vertices_[5] = y - position_height;
544 position_vertices_[8] = x + position_width;
545 position_vertices_[9] = y - position_height;
546 position_vertices_[12] = x + position_width;
547 position_vertices_[13] = y + position_height;
548
549 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, vertex_buffer_);
550 GLsizeiptr vert_size = static_cast<GLsizeiptr>(static_cast<size_t>(16 * sizeof(float)));
551 open_gl_context.extensions.glBufferData(GL_ARRAY_BUFFER, vert_size,
552 position_vertices_, GL_STATIC_DRAW);
553
554 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, triangle_buffer_);
555 position_texture_.bind();
556 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
557 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
558 background_.shader()->use();
559
560 if (background_.texture_uniform() != nullptr)
561 background_.texture_uniform()->set(0);
562
563 background_.enableAttributes(open_gl_context);
564 glDrawElements(GL_TRIANGLES, 6, GL_UNSIGNED_INT, 0);
565 background_.disableAttributes(open_gl_context);
566
567 position_texture_.unbind();
568
569 open_gl_context.extensions.glBindBuffer(GL_ARRAY_BUFFER, 0);
570 open_gl_context.extensions.glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, 0);
571}
